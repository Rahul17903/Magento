<script>
require([
    'jquery'
], function ($) {
    'use strict';

    var endDate = new Date('2026-09-30T23:59:59').getTime();

    function updateCountdown() {
        try {
            var now = new Date().getTime();
            var remaining = endDate - now;

            if (remaining <= 0) {
                $('#ramadan-days').text('00');
                $('#ramadan-hours').text('00');
                $('#ramadan-minutes').text('00');
                $('#ramadan-seconds').text('00');

                return;
            }

            var days = Math.floor(remaining / (1000 * 60 * 60 * 24));

            var hours = Math.floor(
                (remaining % (1000 * 60 * 60 * 24)) /
                (1000 * 60 * 60)
            );

            var minutes = Math.floor(
                (remaining % (1000 * 60 * 60)) /
                (1000 * 60)
            );

            var seconds = Math.floor(
                (remaining % (1000 * 60)) /
                1000
            );

            $('#ramadan-days').text(String(days).padStart(2, '0'));
            $('#ramadan-hours').text(String(hours).padStart(2, '0'));
            $('#ramadan-minutes').text(String(minutes).padStart(2, '0'));
            $('#ramadan-seconds').text(String(seconds).padStart(2, '0'));

        } catch (error) {
            console.error('Countdown error:', error);
        }
    }

    updateCountdown();

    setInterval(updateCountdown, 1000);
});
</script>
